cmake_minimum_required(VERSION 3.24)
project(execution-consumer LANGUAGES CXX)
enable_testing()
include("${CMAKE_CURRENT_LIST_DIR}/../RuntimeDlls.cmake")
set(CMAKE_DISABLE_FIND_PACKAGE_stlab TRUE)
find_package(stlab-execution REQUIRED CONFIG)
# Package discovery must be repeatable, including its platform dependencies.
find_package(stlab-execution REQUIRED CONFIG)

get_target_property(imported stlab::execution IMPORTED)
if(NOT imported OR TARGET execution OR TARGET stlab::stlab OR TARGET stlab::stlab-core)
  message(FATAL_ERROR "Standalone installed consumption requires only imported stlab::execution.")
endif()
get_target_property(headers stlab::execution HEADER_SET_headers)
get_target_property(includes stlab::execution INTERFACE_INCLUDE_DIRECTORIES)
if(EXECUTION_PREFIX)
  file(REAL_PATH "${EXECUTION_PREFIX}" prefix)
  foreach(directory IN LISTS includes)
    string(REGEX REPLACE "^\\$<BUILD_INTERFACE:(.*)>$" "\\1" directory "${directory}")
    file(REAL_PATH "${directory}" directory)
    cmake_path(IS_PREFIX prefix "${directory}" NORMALIZE installed)
    if(NOT installed)
      message(FATAL_ERROR "Execution headers must come only from the install: ${directory}")
    endif()
  endforeach()
endif()
file(WRITE "${CMAKE_BINARY_DIR}/package-evidence.txt"
  "target=stlab::execution\nimported=${imported}\nincludes=${includes}\nheaders=${headers}\n")
if(NOT headers)
  message(FATAL_ERROR "The installed execution header file set is missing.")
endif()

add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE stlab::execution)
execution_copy_runtime_dlls(consumer)
add_test(NAME execution.package.consumer COMMAND consumer)
set_tests_properties(execution.package.consumer PROPERTIES TIMEOUT 10)

set(header_index 0)
foreach(header IN LISTS headers)
  if(NOT EXISTS "${header}")
    message(FATAL_ERROR "Missing installed header: ${header}")
  endif()
  string(REGEX REPLACE "^.*[/\\\\]include[/\\\\]" "" canonical "${header}")
  if(canonical MATCHES "^stlab/concurrency/detail/")
    if(NOT TARGET libdispatch::libdispatch)
      message(STATUS "Platform-specific header not applicable: ${canonical}")
      continue()
    endif()
  endif()
  math(EXPR header_index "${header_index} + 1")
  file(WRITE "${CMAKE_BINARY_DIR}/h${header_index}.cpp" "#include <${canonical}>\n")
  add_library(header.${header_index} OBJECT "${CMAKE_BINARY_DIR}/h${header_index}.cpp")
  target_link_libraries(header.${header_index} PRIVATE stlab::execution)
endforeach()
